export async function discount(tx, id, lowered) {
  const data = { pricePiasters: lowered };
  await tx.package.updateMany({ where: { id }, data });
  await tx.package.createMany({ data: [{ ...(lowered === null ? {} : { pricePiasters: lowered }) }] });
}
